# LiteAI - Raspberry Pi firmware configuration.
#
# Read by the VideoCore firmware before Linux starts. Keep this file on the
# FAT boot partition; the Pi cannot boot without it.

[all]
# 64-bit kernel, built by Arch Linux ARM's linux-rpi package.
arm_64bit=1
kernel=kernel8.img
initramfs initramfs-linux.img followkernel

# Serial console stays available: on a machine whose only interface is a
# terminal, a serial cable is the recovery path of last resort. The HDMI
# text console needs no KMS overlay - the firmware framebuffer already
# provides one - and skipping vc4-kms-v3d leaves the RAM to the model.
enable_uart=1

[pi4]
# Higher clocks on Pi 4 once the firmware has confirmed adequate cooling.
arm_boost=1

[pi5]
arm_boost=1
